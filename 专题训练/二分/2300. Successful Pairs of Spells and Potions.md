```python
def lower_bound(potions: List[int],target: int) -> int:
    left = 0
    right = len(potions) - 1

    while left <= right:
        mid = (left+right) // 2
        if potions[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return left

class Solution:
    def successfulPairs(self, spells: List[int], potions: List[int], success: int) -> List[int]:
        n = len(spells)
        aim = [success]*n
        amt = [0]*n
        potions.sort()
        for i in range(n):
            aim[i] = ceil(success/ spells[i])
            amt[i] = len(potions) - lower_bound(potions,aim[i])
        return amt
```

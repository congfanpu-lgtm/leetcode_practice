def lower_bound(arr2: List[int], target) -> int:
    left = 0
    right = len(arr2) - 1

    while left <= right:
        mid = (left + right) // 2
        if arr2[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return left

class Solution:
    def findTheDistanceValue(self, arr1: List[int], arr2: List[int], d: int) -> int:
        amount = 0
        arr2.sort()
        for x in arr1:
            index = lower_bound(arr2, x)
            if index == 0:
                dis = abs(x - arr2[index])
            elif index == len(arr2):
                dis = abs(x - arr2[index-1])
            else:
                a = abs(x - arr2[index])
                b = abs(x - arr2[index-1])
                dis = min(a,b)
            if dis > d:
                amount += 1
        return amount

```python
def lower_bound(nums: List[int], target: int) -> int:
    left = 0
    right = len(nums) -1
    while left <= right:
        mid = (left + right) // 2
        if nums[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return left

class Solution:
    def maximumCount(self, nums: List[int]) -> int:
        amtp = len(nums) - lower_bound(nums,1)
        amtn = lower_bound(nums,0)
        return max(amtp, amtn)
```

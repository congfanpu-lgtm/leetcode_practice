```python
class Solution:
    def restoreIpAddresses(self, s: str) -> List[str]:
        n = len(s)
        ans = []
        path = [0] * 4
  
        def dfs(i:int, j:int , ip_val:int) -> None:
            if i == n:
                if j == 4:
                    a, b, c, _ = path
                    ans.append(f"{s[:a]}.{s[a:b]}.{s[b:c]}.{s[c:]}")
                return
            if j == 4:
                return
  
            ip_val = ip_val * 10 + int(s[i])
            if ip_val>255:
                return
  
            if ip_val>0:
                dfs(i+1,j,ip_val)
  
            path[j] = i+1
            dfs(i+1,j+1,0)
        dfs(0,0,0)
        return ans
```

# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def reversehead(self, head:Optional[ListNode]) -> Optional[ListNode]:
        pre = None
        cur = head
        while cur:
            nxt = cur.next
            cur.next = pre
            pre = cur
            cur = nxt
        return pre

    def removeNthFromEnd(self, head: Optional[ListNode], n: int) -> Optional[ListNode]:
        head1 = self.reversehead(head)
        dummy = ListNode(0,head1)
        pre = dummy
  
        for _ in range(n-1):
            pre = pre.next
        if pre.next is not None and pre.next.next is not None:
            pre.next = pre.next.next
        else:
            pre.next = None
            
        return self.reversehead(dummy.next)
```
**注意**：
1. 注意加哨兵节点，加哨兵方便处理删除头节点的情况

# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def removeNthFromEnd(self, head: Optional[ListNode], n: int) -> Optional[ListNode]:
        dummy = ListNode(next=head)
        right = dummy

        for _ in range(n):
            right = right.next

        left = dummy
        while right.next:
            left = left.next
            right = right.next
        left.next = left.next.next
        return dummy.next
```
**方法**：
1. 采取相对距离来实现

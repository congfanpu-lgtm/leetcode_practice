```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next

class Solution:
    def removeNthFromEnd(self, head: Optional[ListNode], n: int) -> Optional[ListNode]:
        nn = 0
        cur = head
        while cur:
            nn += 1
            cur = cur.next
        cur = head
        for _ in range(nn - n - 1):
            cur = cur.next
        if n == 1:
            cur.next = None
        else:
            cur.next = cur.next.next
        return cur
```

class Solution:
    def trap(self, height: List[int]) -> int:
        n = len(height)
        l_height = [0] * n
        r_height = [0] * n
        l = 0
        for i,x in enumerate(height):
            l = max(l,x)
            l_height[i] = l
        r = 0
        for i in range(n-1,-1,-1):
            x = height[i]
            r = max(r,x)
            r_height[i] = r
        ans = 0
        for i,x in enumerate(height):
            ans = ans + min(l_height[i],r_height[i]) - x
        return ans

class Solution:
    def trap(self, height: List[int]) -> int:
        n = len(height)
        pre_max = [0] * n  # pre_max[i] 表示从 height[0] 到 height[i] 的最大值
        pre_max[0] = height[0]
        for i in range(1, n):
            pre_max[i] = max(pre_max[i - 1], height[i])
  
        suf_max = [0] * n  # suf_max[i] 表示从 height[i] 到 height[n-1] 的最大值
        suf_max[-1] = height[-1]
        for i in range(n - 2, -1, -1):
            suf_max[i] = max(suf_max[i + 1], height[i])
  
        ans = 0
        for h, pre, suf in zip(height, pre_max, suf_max):
            ans += min(pre, suf) - h  # 累加每个水桶能接多少水
        return ans
```

class Solution:
    def trap(self, height: List[int]) -> int:
        ans = 0
        st = []
        for i, h in enumerate(height):
        # 当当前柱子比栈顶柱子高时，说明可以形成凹槽
            while st and h >= height[st[-1]]:
                bottom_h = height[st.pop()] # 弹出栈顶作为凹槽底部
                # 如果栈空了，说明左边没有柱子了，无法形成凹槽
                if len(st) == 0:
                    break
                left = st[-1]
                dh = min(height[left], h) - bottom_h
                ans += dh * (i - left - 1)
            st.append(i)
        return ans
```

**做法一：**
```python
class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        ans = []
        path = []
        n = len(nums)
        def dfs(i):
            if i == n:
                ans.append(path.copy()) # 因为path是全局变量，如果不用copy的话所有的值都会是一样
                return
            # 分支1：不选当前元素 nums[i]
            dfs(i+1)
            # 分支2：选当前元素 nums[i]
            path.append(nums[i])
            dfs(i+1)
            path.pop() # 恢复现场
  
        dfs(0)
        return ans
        
        # 时间复杂度 O(n*2^n)
```

**做法二：**
```python
class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        ans = []
        path = []
        n = len(nums)
        def dfs(i):
            ans.append(path.copy())
            if i == n:
                return
  
            for j in range(i, n):
                path.append(nums[j])
                dfs(j+1)
                path.pop()
  
        dfs(0)
        return ans
```

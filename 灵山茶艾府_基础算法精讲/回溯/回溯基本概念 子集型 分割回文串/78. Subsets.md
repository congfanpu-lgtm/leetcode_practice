做法一：
```python
class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        ans = []
        path = []
        n = len(nums)
        def dfs(i):
            if i == n:
                ans.append(path.copy()) # 因为path是全局变量，如果不用copy的话所有的值都会是一样
                return
            # 分支1：不选当前元素 nums[i]
            dfs(i+1)
            # 分支2：选当前元素 nums[i]
            path.append(nums[i])
            dfs(i+1)
            path.pop()
  
        dfs(0)
        return ans
        
        # 时间复杂度 O(n*2^n)
```

**例一：**
``` 
n=3, nums=[1,2,3], ans=[], path=[]

调用 dfs(0):
    i=0, nums[0]=1
    
    分支1: 不选1
        dfs(1):
            i=1, nums[1]=2
            
            分支1: 不选2
                dfs(2):
                    i=2, nums[2]=3
                    
                    分支1: 不选3
                        dfs(3): i==3 → ans.append([])
                        path=[] → ans=[[]]
                    分支2: 选3
                        path.append(3) → path=[3]
                        dfs(3): i==3 → ans.append([3])
                        path.pop() → path=[]
            
            分支2: 选2
                path.append(2) → path=[2]
                dfs(2):
                    i=2, nums[2]=3
                    
                    分支1: 不选3
                        dfs(3): i==3 → ans.append([2])
                        path=[2] → ans=[[],[3],[2]]
                    分支2: 选3
                        path.append(3) → path=[2,3]
                        dfs(3): i==3 → ans.append([2,3])
                        path.pop() → path=[2]
                path.pop() → path=[]
    
    分支2: 选1
        path.append(1) → path=[1]
        dfs(1):
            i=1, nums[1]=2
            
            分支1: 不选2
                dfs(2):
                    i=2, nums[2]=3
                    
                    分支1: 不选3
                        dfs(3): i==3 → ans.append([1])
                        path=[1] → ans=[[],[3],[2],[2,3],[1]]
                    分支2: 选3
                        path.append(3) → path=[1,3]
                        dfs(3): i==3 → ans.append([1,3])
                        path.pop() → path=[1]
            
            分支2: 选2
                path.append(2) → path=[1,2]
                dfs(2):
                    i=2, nums[2]=3
                    
                    分支1: 不选3
                        dfs(3): i==3 → ans.append([1,2])
                        path=[1,2] → ans=[[],[3],[2],[2,3],[1],[1,3],[1,2]]
                    分支2: 选3
                        path.append(3) → path=[1,2,3]
                        dfs(3): i==3 → ans.append([1,2,3])
                        path.pop() → path=[1,2]
                path.pop() → path=[1]
        path.pop() → path=[]
```


做法二：
```python
class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        ans = []
        path = []
        n = len(nums)
        def dfs(i):
            ans.append(path.copy())
            if i == n:
                return
  
            for j in range(i, n):
                path.append(nums[j])
                dfs(j+1)
                path.pop()
  
        dfs(0)
        return ans
```

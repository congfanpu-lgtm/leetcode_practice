##  参考答案：
---
```python
MAPPING = ["","","abc","def","ghi","jkl","mno","pqrs","tuv","wxyz"]
  
class Solution:
    def letterCombinations(self, digits: str) -> List[str]:
        n = len(digits)
        if n == 0:
            return []
  
        ans = []
        path = [''] * n
        def dfs(i):
            if i == n:
                ans.append(''.join(path))
                return
            for c in MAPPING[int(digits[i])]:
                path[i] = c
                dfs(i+1)
        dfs(0)
        return ans
#时间复杂度O(n*4^n) ans.append生成答案需要O(n),一个数字最多代表4个字母: O(4^n)
```

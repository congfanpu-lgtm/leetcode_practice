```python
## 0-1背包代码
def zero_one_knapsack(capacity: int, w: List[int], v: List[int]) -> int:
    n = len(w)
    def dfs(i,c):
        if i < 0:
            return 0
        if c < w[i]:
            return dfs(i-1, c)
        return max(dfs(i-1,c), dfs(i-1, c - w[i]) + v[i])
    return dfs(n-1, capacity)

  
class Solution:
    def findTargetSumWays(self, nums: List[int], target: int) -> int:
        # p 取p个1
        # s-p
        # p-(s-p) = t
        # 2p=s+t
        # p = (s+t) / 2
        target += sum(nums)
        if target < 0 or target % 2:
            return 0
        target //= 2
        n = len(nums)
  
        @cache
        def dfs(i,c):
            if i < 0:
                return 1 if c == 0 else 0 ## 等于0，则说明正好满足条件，返回1,
            if c < nums[i]:
                return dfs(i-1, c)
            return dfs(i-1,c) + dfs(i-1, c - nums[i])
        return dfs(n-1, target)
```
**注意**：
1. 这里运用加法原理：return dfs(i-1,c) + dfs(i-1, c - nums[i])（总方案数等于两个互斥事件的方案数之和） 
2. 时间复杂度是O(n* p) （状态个数 n * p ) * 每个状态所需时间O(1)

做法二：递推
```python
class Solution:
    def findTargetSumWays(self, nums: List[int], target: int) -> int:
        # p 取p个1
        # s-p
        # p-(s-p) = t
        # 2p=s+t
        # p = (s+t) / 2
        target += sum(nums)
        if target < 0 or target % 2:
            return 0
        target //= 2
        n = len(nums)
  
        f = [[0] * (target + 1) for _ in range(n + 1)] ## 创建n+1行，target+1列数据，包含0，n和0，target
        f[0][0] = 1## 边界条件，到边界时刚好capacity也为0，则返回1
  
        for i,x in enumerate(nums): ##这里nums，是因为下面循环中有i+1
            for c in range(target+1):
                if c < x :
                    f[i+1][c] = f[i][c]  
                else:
                    f[i+1][c] = f[i][c] + f[i][c-x]
        return f[n][target]
```

```python
class Solution:
    def findTargetSumWays(self, nums: List[int], target: int) -> int:
        # p 取p个1
        # s-p
        # p-(s-p) = t
        # 2p=s+t
```

```python
        n = len(nums)
  
        f = [0] * (target + 1)
        f[0] = 1
  
        for x in nums:
            for c in range(target, x-1, -1):
                    f[c] = f[c] + f[c-x]
        return f[target]

```

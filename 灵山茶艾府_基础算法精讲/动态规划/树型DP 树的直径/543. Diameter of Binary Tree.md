## 回答（错误）：
---
```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def deep(self, root: Optional[TreeNode]) -> int:
        ans = 0
        def f(node, cnt):
            if node is None:
                return
            cnt += 1
            nonlocal ans
            ans = max(ans, cnt)
            f(node.left,cnt)
            f(node.right,cnt)
        f(root,0)
        return ans
    def diameterOfBinaryTree(self, root: Optional[TreeNode]) -> int:
        left = self.deep(root.left)
        right = self.deep(root.right)
        ans = left + right
        return ans
```
**错误原因**：
1. 可能不经过根节点，如果不经过根节点的直径，则无法用“左子树深度 + 右子树深度"来计算直径
## 参考答案：
---
```python
class Solution:
    def diameterOfBinaryTree(self, root: Optional[TreeNode]) -> int:
        ans = 0
        def dfs(node):
            if node is None:
                return -1
            # l_len = dfs(node.left)
            # r_len = dfs(node.right)
            # nonlocal ans
            # ans = max(ans, l_len + r_len + 2)
            # return max(l_len, r_len) + 1
            l_len = dfs(node.left) + 1
            r_len = dfs(node.right) + 1
            nonlocal ans
            ans = max(ans, l_len + r_len)
            return max(l_len, r_len)
        dfs(root)
        return ans
```
时间复杂度和空间复杂度为O(n)

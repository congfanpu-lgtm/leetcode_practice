class Solution:
    def minSubArrayLen(self, target: int, nums: List[int]) -> int:
        n = len(nums)
        amount = 0
        res = 0

        if sum(nums) < target:
            res = 0
        elif target in nums
            res = 1
        else:
            num = 0

            for i in range(n-1):
                temp_amount = nums[i]
                for j in range(i+1,n):
                    temp_amount = temp_amount + nums[j]
                    
                    if temp_amount < target:
                        continue
                    elif temp_amount == target:
                        temp_num = j - i + 1
                        
                        if num == 0 and temp_num > 0:
                            num = temp_num
                        elif temp_num > 0 and temp_num < num:
                            num = temp_num
                    else:
                        break
            res = num
        return res

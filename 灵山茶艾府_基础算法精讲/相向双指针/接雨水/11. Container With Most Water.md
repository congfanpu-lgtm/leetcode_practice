```python
class Solution:
    def maxArea(self, height: List[int]) -> int:
        n = len(height)
        res = 0

        for l in range(n):
            for r in range(l + 1, n):
                tempres = height[l]*(r-l) if height[l] < height[r] else height[r]*(r-l)
                res = res if res > tempres else tempres
        return res
```

```python
class Solution:
    def maxArea(self, height: List[int]) -> int:
        n = len(height)
        res = 0
        l = 0
        r = n-1

        while l < r:
            temp_res = min(height[l],height[r])*(r-l)
            res = temp_res if temp_res > res else res
            if height[l] < height[r]:
                l += 1
            else:
                r -= 1
        return res
```

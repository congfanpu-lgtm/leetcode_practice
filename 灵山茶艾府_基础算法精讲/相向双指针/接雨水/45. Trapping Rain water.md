```python
class Solution:
    def trap(self, height: List[int]) -> int:
        n = len(height)
        maxh = 0
        amount = 0
        arr1 = list(range(n))
        arr2 = list(range(n))
        
        for i in range(n): ##左最大
            maxh = max(maxh,height[i])
            arr1[i] = maxh
        ##右最大
        maxh = 0 
        for j in range(n-1,-1,-1):
            maxh = max(maxh,height[j])
            arr2[j] = maxh
        ##合并
        arr3 = list(range(n))
        for k in range(n):
            arr3[k] = min(arr1[k],arr2[k])
        ##遍历计算
        for m in range(n):
            add = arr3[m] - height[m] if arr3[m] > height[m] else 0
            amount += add

        return amount
```

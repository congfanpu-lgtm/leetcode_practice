**灵茶（夯爆了）**：
```python
##要求 nums是非递减的，即nums[i] <= nums[i+1]
##返回最小的满足 nums[i] >= target 的 i
##如果不存在，返回 len(nums)
def lower_bound(nums: List[int], target: int) -> int:
    #时间复杂度Olog(n)
    #空间复杂度O(1)
    left = 0
    right = len(nums) - 1 # 闭区间[left, right]
    while left <= right: # 区间不为空
        #left + (right - left)/2 防止其他语言溢出  
        mid = (left + right) // 2
        if nums[mid] < target:
            left = mid + 1 #[mid+1, right]
        else:
            right = mid - 1 #[left,mid -1]
    return left

def lower_bound2(nums: List[int], target: int) -> int:
    left = 0
    right = len(nums) # 左闭右开区间[left, right)
    while left < right: # 区间不为空
        #left + (right - left)/2 防止其他语言溢出  
        mid = (left + right) // 2
        if nums[mid] < target:
            left = mid + 1 #[mid+1, right)
        else:
            right = mid #[left,mid)
    return left # right

def lower_bound3(nums: List[int], target: int) -> int:
    left = -1
    right = len(nums) # 开区间(left, right)
    while left + 1 < right: # 区间不为空
        #left + (right - left)/2 防止其他语言溢出  
        mid = (left + right) // 2
        if nums[mid] < target:
            left = mid #(mid+1, right)
        else:
            right = mid #(left,mid)
    return right

class Solution:
    def searchRange(self, nums: List[int], target: int) -> List[int]:
        start = lower_bound(nums, target)
        if start == len(nums) or nums[start] != target:
            return [-1, -1]
        end = lower_bound(nums, target + 1) -1## end的计算是用了下面函数定义中第一个大于target的数-1就是最后一个=target的脚标 
        return [start, end] 
```

定义函数解释：有三种方法，并且对于 >=, >, < <=可以进行转换

```python
def mini_index(nums:List[int]) -> int:
    left = 0
    n = len(nums) - 1
    right = n
    while left <= right:
        mid = (left + right) // 2
        if nums[mid] > nums[n]:
            left = mid + 1
        else:
            right = mid - 1
    return left

class Solution:
    def findMin(self, nums: List[int]) -> int:
        num = nums[mini_index(nums)]
        return num
```

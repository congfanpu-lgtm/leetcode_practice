```python
def findmini(nums: List[int]) -> int:
    left = 0
    right = len(nums) - 1
    while left <= right:
        mid = (right +left) // 2
        if nums[mid] > nums[-1]:
            left = mid + 1
        else:
            right = mid - 1
    return left
  

class Solution:
    def search(self, nums: List[int], target: int) -> int:
        mini = findmini(nums)
        n = len(nums)
        if target > nums[-1]:
            arr = nums[0:mini-1]
            index = bisect_left(arr,target)
            if nums[index] != target:
                index = -1
        else:
            arr = nums[mini:n-1]
            index = bisect_left(arr,target)+mini
            if nums[index] != target:
                index = -1
        return index
```
思路：两次二分，第一次找最小值，第二次找目标值，通过判断是否比最后一个数大来决定在左边区间还是右边区间来找

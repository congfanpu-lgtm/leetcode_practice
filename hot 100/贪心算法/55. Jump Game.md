```python
class Solution:
    def canJump(self, nums: List[int]) -> bool:
        n = len(nums)
        k = 1
        for _ in range(n):
            if n == 1:
                return True
            if nums[n - 1 - k ] >= k:
                n = n - k
                k = 1
            else:
                k += 1
        return False
```

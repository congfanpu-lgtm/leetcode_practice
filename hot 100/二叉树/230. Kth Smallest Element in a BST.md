```python
class Solution:
    def kthSmallest(self, root: Optional[TreeNode], k: int) -> int:
        sum = 0
        while root:
            if root.left:
                pre = root.left
                while pre.right and pre.right is not root:
                    pre = pre.right
                if pre.right is None:
                    pre.right = root
                    root = root.left
                    continue
  
                pre.right = None
            sum += 1
            if sum == k:
                return root.val
            root = root.right
```

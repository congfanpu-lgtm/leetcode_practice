```python
class Solution:
    def numSquares(self, n: int) -> int:
        ans = inf
        nums = []
        for i in range(1,n+1):
            if i ** 2 > n:
                break
            nums.append(i**2)
        t = len(nums)
        @cache
        def dfs(i,c):
            if i < 0:
                return 0 if c == 0 else inf
            if c < nums[i]:
                return dfs(i-1,c)
            return min(dfs(i-1,c),dfs(i,c-nums[i])+1)

        return dfs(t-1,n)
```

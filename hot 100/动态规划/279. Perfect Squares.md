## 回答(超出内存限制）：

```python
class Solution:
    def numSquares(self, n: int) -> int:
        ans = inf
        nums = []
        for i in range(1,n+1):
            if i ** 2 > n:
                break
            nums.append(i**2)
        t = len(nums)
        @cache
        def dfs(i,c):
            if i < 0:
                return 0 if c == 0 else inf
            if c < nums[i]:
                return dfs(i-1,c)
            return min(dfs(i-1,c),dfs(i,c-nums[i])+1)

        return dfs(t-1,n)
```

## 参考答案：
---
记忆化搜索：
```python
# 写在外面，多个测试数据之间可以共享，减少计算量
@cache  # 缓存装饰器，避免重复计算 dfs 的结果（记忆化）
def dfs(i: int, j: int) -> int:
    if i == 0:
        return inf if j else 0
    if j < i * i:
        return dfs(i - 1, j)  # 只能不选
    return min(dfs(i - 1, j), dfs(i, j - i * i) + 1)  # 不选 vs 选
  
class Solution:
    def numSquares(self, n: int) -> int:
        return dfs(isqrt(n), n)
```

递推：
```python
N = 10000
f = [[0] * (N + 1) for _ in range(isqrt(N) + 1)]
f[0] = [0] + [inf] * N
for i in range(1, len(f)):
    for j in range(N + 1):
        if j < i * i:
            f[i][j] = f[i - 1][j]  # 只能不选
        else:
            f[i][j] = min(f[i - 1][j], f[i][j - i * i] + 1)  # 不选 vs 选
  
class Solution:
    def numSquares(self, n: int) -> int:
        return f[isqrt(n)][n]  # 也可以写 f[-1][n]
```

空间优化
```python
N = 10000
f = [0] + [inf] * N
for i in range(1, isqrt(N) + 1):
    for j in range(i * i, N + 1):
        f[j] = min(f[j], f[j - i * i] + 1)  # 不选 vs 选
  
class Solution:
    def numSquares(self, n: int) -> int:
        return f[n]
```

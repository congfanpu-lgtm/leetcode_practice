```python
class Solution:
    def canPartition(self, nums: List[int]) -> bool:
        n = sum(nums)
        m = len(nums)
        if n % 2:
            return False
        @cache
        def dfs(i,c) ->bool:
            if i >= m or c < 0:
                return False
            if c == n/2:
                return True
            return dfs(i+1, c-nums[i]) or dfs(i+1, c) ## 只有有一种方法可以分割，那么返回正确，负责错误
        return dfs(0,n)
```

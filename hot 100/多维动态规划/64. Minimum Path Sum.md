```python
class Solution:
    def minPathSum(self, grid: List[List[int]]) -> int:
        m,n = len(grid),len(grid[0])
        @cache  # 缓存装饰器，避免重复计算 dfs 的结果（一行代码实现记忆化）
        def dfs(i: int, j: int) -> int:
            if i > m-1 or j > n-1:
                return inf
            if i == m-1 and j == n-1:
                return grid[i][j]
            return min(dfs(i + 1, j), dfs(i, j + 1)) +grid[i][j]
        return dfs(0,0)
```
时间复杂度O(mn),空间复杂度O(mn)?
